# 统一的陶土橙渐变(绝对坐标, 让 12 片星芒共用同一束光)
function New-MarkBrush {
    $b = New-Object System.Windows.Media.RadialGradientBrush
    $b.MappingMode = [System.Windows.Media.BrushMappingMode]::Absolute
    $b.Center = New-Object System.Windows.Point(120, 92)
    $b.GradientOrigin = New-Object System.Windows.Point(120, 92)
    $b.RadiusX = 152; $b.RadiusY = 152
    [void]$b.GradientStops.Add((New-Gs '#FFFFC65C' 0))
    [void]$b.GradientStops.Add((New-Gs '#FFFF7AAE' 0.45))
    [void]$b.GradientStops.Add((New-Gs '#FF8B5CF6' 1))
    return $b
}

# 十二道等宽圆头射线: 和中间的圆盘合起来就是 Apple 星号(整只就是标志轮廓)
# 局部坐标: (0,0) = 宠物中心; 齿从半径 26 长到 94, 所以转起来是绕中心
$script:PetalData = 'M -9.5,-26.5 L -9.5,-84.5 A 9.5,9.5 0 0 1 9.5,-84.5 L 9.5,-26.5 A 9.5,9.5 0 0 1 -9.5,-26.5 Z'
function Initialize-Blossom {
    for ($i = 0; $i -lt 0; $i++) {
        $p = New-Object System.Windows.Shapes.Path
        $p.Data = [System.Windows.Media.Geometry]::Parse($script:PetalData)
        $p.Fill = New-MarkBrush
        $p.RenderTransformOrigin = New-Object System.Windows.Point(0, 0)
        $p.RenderTransform = New-Object System.Windows.Media.RotateTransform($i * 30)
        $p.IsHitTestVisible = $false
        [System.Windows.Controls.Canvas]::SetLeft($p, $script:CenterX)
        [System.Windows.Controls.Canvas]::SetTop($p, $script:CenterY)
        [void]$BlossomLayer.Children.Add($p)
    }
}
Initialize-Blossom

# 身边飘的三颗小闪光
$script:Orbit = New-Object System.Collections.ArrayList
# Apple Intelligence 标志: 7 个白色环互相交叠成一圈花结(照老图标)
function Initialize-Orbit {
    for ($i = 0; $i -lt 0; $i++) {
        $a = 2 * [Math]::PI * $i / 7
        $cx = 120 + [Math]::Cos($a) * 30
        $cy = 122 + [Math]::Sin($a) * 30
        $p = New-Object System.Windows.Shapes.Path
        $p.Data = [System.Windows.Media.Geometry]::Parse(('M {0:0.##},{1:0.##} A 34,50 0 1 1 {2:0.##},{1:0.##} A 34,50 0 1 1 {3:0.##},{1:0.##} Z' -f ($cx - 34), $cy, ($cx + 34), ($cx - 34)))
        $p.Fill = $null
        $p.Stroke = New-Object System.Windows.Media.SolidColorBrush([System.Windows.Media.Color]::FromArgb(255, 255, 255, 255))
        $p.StrokeThickness = 13
        $p.StrokeStartLineCap = 'Round'
        $p.StrokeEndLineCap = 'Round'
        $p.RenderTransformOrigin = New-Object System.Windows.Point(0.5, 0.5)
        $p.RenderTransform = New-Object System.Windows.Media.RotateTransform(($i * 360.0 / 7.0 + 12), 120, 122)
        $p.IsHitTestVisible = $false
        [void]$OrbitLayer.Children.Add($p)
    }
}
Initialize-Orbit {
    $ring = New-Object System.Windows.Shapes.Path
    $ring.Data = [System.Windows.Media.Geometry]::Parse('M 30,122 A 90,90 0 1 0 210,122 A 90,90 0 1 0 30,122 Z M 62,122 A 58,58 0 1 0 178,122 A 58,58 0 1 0 62,122 Z')
    $ring.Fill = New-LinBrush -colors @('#FF2A2A33', '#FF241033') -offsets @(0, 1)
    $ring.IsHitTestVisible = $false
    [void]$OrbitLayer.Children.Add($ring)
    $slash = New-Object System.Windows.Shapes.Path
    $slash.Data = [System.Windows.Media.Geometry]::Parse('M 176,44 L 196,60 L 64,200 L 44,184 Z')
    $slash.Fill = New-LinBrush -colors @('#FF2A2A33', '#FF241033') -offsets @(0, 1)
    $slash.IsHitTestVisible = $false
    [void]$BlossomLayer.Children.Add($slash)
}
Initialize-Orbit {
    $i = 0
    foreach ($phase in @(0.6, 2.7, 4.8)) {
        $p = New-Object System.Windows.Shapes.Path
        $p.Data = [System.Windows.Media.Geometry]::Parse($script:SparkData)
        $p.Fill = New-LinBrush -colors @('#F2FFFFFF', '#FFFFB05C') -offsets @(0, 1)
        $p.RenderTransformOrigin = New-Object System.Windows.Point(0, 0)
        $rt = New-Object System.Windows.Media.RotateTransform(0)
        $p.RenderTransform = $rt
        $p.IsHitTestVisible = $false
        [void]$OrbitLayer.Children.Add($p)
        [void]$script:Orbit.Add(@{
                El = $p; Rot = $rt; Phase = $phase
                Speed = 0.30 + 0.06 * $i
                Radius = 92.0 + 8.0 * $i
            })
        $i++
    }
}
Initialize-Orbit
